#!/usr/bin/env python3
"""classify.py <capture-file>: ACCEPT only a NEW, FINAL reply with pi settled idle. Prints JSON; exit 0 accept, 1 refuse."""
import json, re, sys
BUSY = re.compile(r"working|esc to interrupt|[⠋⠙⠹⠸⠼⠴⠦⠧⠇⠏]", re.I)
def classify(text):
    lines = [l.rstrip() for l in text.splitlines()]
    if not any(l.strip() for l in lines):
        return False, "empty capture", None
    starts = [i for i, l in enumerate(lines) if "Call only mcp__" in l]
    if not starts:
        return False, "prompt echo not found", None
    body = lines[starts[-1] + 1:]
    # pi idle layout: region, then separator rule(s), cwd line, footer line at the bottom
    try:
        sep = next(i for i, l in enumerate(body) if l.lstrip().startswith("─"))
    except StopIteration:
        return False, "no input-box separator: pi not settled", None
    region = [l.strip() for l in body[:sep] if l.strip()]
    tail = [l for l in body[sep:] if l.strip()]
    footer = [l for l in tail if "(sub)" in l]
    if any(BUSY.search(l) for l in lines):
        return False, "busy indicator present: unfinished", None
    if not footer:
        return False, "no footer: pi not settled", None
    if not re.search(r"↑\d+", footer[-1]):
        return False, "stale footer: no token counter, no new reply since the prompt", None
    if not region:
        return False, "no reply after the prompt", None
    tool_idx = [i for i, l in enumerate(region) if "mcp__cpp_pi1899_" in l]
    final = region[tool_idx[-1] + 1:] if tool_idx else region
    final = [l for l in final if "mcp__cpp_pi1899_" not in l]
    if not final:
        return False, "tool-call phase: no final answer after the tool event", None
    return True, "final reply, pi idle", {"tool_event_lines": len(tool_idx), "reply": "\n".join(final)}
if __name__ == "__main__":
    ok, why, data = classify(open(sys.argv[1], encoding="utf-8", errors="replace").read())
    print(json.dumps({"accept": ok, "reason": why, **(data or {})}, ensure_ascii=False))
    sys.exit(0 if ok else 1)
