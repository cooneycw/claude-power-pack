#!/bin/bash
# record.sh <tag> <transport>: record the ACCEPTED reply, save receipt+hash, THEN delete the pane log. Never deletes first.
T5=$(dirname "$0"); L=$1; TR=$2; A=$T5/accept-$L.json; RC=$T5/record-$L.receipt
[ -s "$A" ] || { echo "REFUSE: no accepted reply for $L; pane log KEPT"; exit 1; }
cd "$T5/cpp"
python3 -c 'import json,sys; sys.stdout.write(json.load(open(sys.argv[1]))["reply"]+"\n")' "$A" \
  | python3 docs/measurements/pi/issue-1899-b3-probe.py record-reply --tag $L --root "$HOME/.cpp-pi-1899-probe" > "$RC" 2>&1; rc=$?
echo "record_exit=$rc" >> "$RC"
F="$HOME/.cpp-pi-1899-probe/work/pi-1899-b3/replies/$L.txt"
[ $rc -eq 0 ] && [ -s "$F" ] || { cat "$RC"; echo "REFUSE: record-reply failed; pane log KEPT"; exit 1; }
sha256sum "$F" >> "$RC"; sync
cat "$RC"
python3 docs/measurements/pi/issue-1899-b3-probe.py verify --tag $L --transport $TR --root "$HOME/.cpp-pi-1899-probe"; echo "verify_exit=$?"
rm -f -- "$T5/pipe-$L.log" && echo "pane log deleted after durable record"
