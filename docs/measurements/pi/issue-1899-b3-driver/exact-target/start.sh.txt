#!/bin/bash
# start.sh <alpha|beta> <tag>: launch, check pin+footer, leave pi running at prompt
T5=$(dirname "$0"); r=$1; L=$2; X=$T5/exit-$L; P=$T5/pipe-$L.log
rm -f -- "$X" "$P"
tmux new-session -d -s pib3 -x 400 -y 100 "bash -l $T5/launch-$r.sh $X"
tmux pipe-pane -t =pib3 -o "cat >> $P"
for i in $(seq 1 60); do tmux capture-pane -p -J -t =pib3 | grep -q '(sub)' && break; sleep 1; done
sleep 2
pin=$(sed 's/\x1b\[[0-9;?]*[a-zA-Z]//g' "$P" | grep -a -o 'Pinned model: [^[:space:]]*' | head -1)
foot=$(tmux capture-pane -p -J -t =pib3 | grep '(sub)' | tail -1 | sed 's/  \+/  /g' | awk -F'  ' '{print $NF}')
echo "PIN_LINE=[$pin] FOOTER=[$foot]"
fm=${foot%% •*}
if [ "$pin" != "Pinned model: github-copilot/gpt-5.3-codex" ] || [ "$fm" != "gpt-5.3-codex" ]; then echo "STOP: pin/footer mismatch"; exit 9; fi
